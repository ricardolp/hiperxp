FINAL = 64  # display size in files; map scales via CSS

def scrub_qty(im: Image.Image) -> Image.Image:
    '''Replace bottom-right quantity digits with nearby background.'''
    im = im.copy()
    w, h = im.size
    px = im.load()
    # sample median-ish from left of qty zone
    samples = []
    for y in range(h*2//3, h-4):
      for x in range(w//2, w*2//3):
        samples.append(px[x,y])
    samples.sort()
    bg = samples[len(samples)//2]
    for y in range(h - h//4, h):
      for x in range(w - w//2, w):
        r,g,b = px[x,y]
        # white/light text or high-contrast digits
        if r > 180 and g > 180 and b > 180:
          px[x,y] = bg
        elif r+g+b > 500:
          px[x,y] = bg
    # also soft-fill the whole qty corner with slight blur of bg patch
    patch = im.crop((w//2, h*2//3, w*3//4, h-2)).resize((w//2, h//4), Image.BILINEAR)
    # only paste where still looks like text remnant: use mask of bright pixels
    corner = im.crop((w//2, h*3//4, w, h))
    mask = Image.new('L', corner.size, 0)
    mp = mask.load()
    cp = corner.load()
    for y in range(corner.size[1]):
      for x in range(corner.size[0]):
        r,g,b = cp[x,y]
        if r>170 and g>170 and b>170:
          mp[x,y] = 255
    # expand mask a bit
    mask = mask.filter(ImageFilter.MaxFilter(3))
    fill = Image.new('RGB', corner.size, bg)
    corner.paste(fill, (0,0), mask)
    im.paste(corner, (w//2, h*3//4))
    return im
